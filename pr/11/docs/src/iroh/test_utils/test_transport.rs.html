52    /// # Example
53    ///
54    /// ```ignore

70/// A simulated network for testing custom transports.
71///
72/// This allows creating multiple [`TestTransport`] instances that can communicate
73/// with each other through in-memory channels.
74///
75/// # Example
76///
77/// ```ignore
78/// use iroh::test_utils::custom_transport::TestNetwork;
79///
80/// let network = TestNetwork::new();
81/// let transport1 = network.create_transport(endpoint_id1)?;
82/// let transport2 = network.create_transport(endpoint_id2)?;
83/// // transport1 and transport2 can now communicate via the network
84/// ```
85#[derive(Debug, Clone, Default)]
86pub struct TestNetwork {
87    inner: Arc<Mutex<TestNetworkInner>>,
88}
89
90impl TestNetwork {
91    /// Creates a new empty test network.
92    pub fn new() -> Self {
93        Self::default()
94    }
95
96    /// Creates an address lookup service for this network.
97    pub fn address_lookup(&self) -> impl AddressLookup {
98        TestAddrLookup {
99            network: self.clone(),
100        }
101    }
102
103    /// Starts or stops dropping every packet carried by this test transport.
104    ///
105    /// While enabled, sends still report success but nothing is delivered —
106    /// exactly like a network path that went dark. Other transports on the
107    /// same endpoints (relay, IP) are unaffected, so this simulates the field
108    /// condition "the direct path is dead while the relay stays healthy".
109    pub fn set_blackhole(&self, blackhole: bool) {
110        self.inner.lock().expect("poisoned").blackhole = blackhole;
111    }
112
113    /// Creates a new test transport for the given endpoint ID.
114    ///
115    /// Returns an error if the ID already exists in the network.
116    pub fn create_transport(&self, id: EndpointId) -> io::Result<Arc<TestTransport>> {
117        let id_custom = to_custom_addr(id);
118        let mut guard = self.inner.lock().expect("poisoned");
119        if guard.channels.contains_key(&id) {
120            return Err(io::Error::other("endpoint ID already exists in network"));
121        }
122        guard.channels.insert(id, mpsc::channel(256));
123        drop(guard);
124        Ok(Arc::new(TestTransport {
125            id_watchable: n0_watcher::Watchable::new(vec![id_custom]),
126            network: self.clone(),
127            id,
128        }))
129    }
130}

147    fn resolve(
148        &self,
149        endpoint_id: EndpointId,
150    ) -> Option<n0_future::stream::Boxed<Result<Item, crate::address_lookup::Error>>> {
151        if self
152            .network
153            .inner
154            .lock()
155            .expect("poisoned")
156            .channels
157            .contains_key(&endpoint_id)
158        {
159            Some(Box::pin(n0_future::stream::once(Ok(Item::new(
160                EndpointInfo {
161                    endpoint_id,
162                    data: EndpointData::from_iter([TransportAddr::Custom(CustomAddr::from_parts(
163                        TEST_TRANSPORT_ID,
164                        endpoint_id.as_bytes(),
165                    ))]),
166                },
167                "test discovery",
168                None,
169            )))))
170        } else {
171            None
172        }
173    }

182/// Converts an endpoint ID to a custom address for this test transport.
183pub fn to_custom_addr(endpoint: EndpointId) -> CustomAddr {
184    CustomAddr::from((TEST_TRANSPORT_ID, &endpoint.as_bytes()[..]))
185}
186
187fn try_parse_custom_addr(addr: &CustomAddr) -> io::Result<EndpointId> {
188    if addr.id() != TEST_TRANSPORT_ID {
189        return Err(io::Error::other("unexpected transport id"));
190    }
191    let key_bytes: &[u8; 32] = addr
192        .data()
193        .try_into()
194        .map_err(|_| io::Error::other("wrong key length"))?;
195    EndpointId::from_bytes(key_bytes).map_err(|_| io::Error::other("KeyParseError"))
196}
197
198impl TestSender {
199    fn send_sync(&self, dst: &CustomAddr, packets: Vec<Packet>) -> io::Result<()> {
200        let to_id = try_parse_custom_addr(dst)?;
201        let guard = self.network.inner.lock().expect("poisoned");
202        if guard.blackhole {
203            info!(
204                "send {} -> {}: dropped {} packets (blackhole)",
205                self.id.fmt_short(),
206                to_id.fmt_short(),
207                packets.len()
208            );
209            return Ok(());
210        }
211        let (s, _) = guard
212            .channels
213            .get(&to_id)
214            .ok_or_else(|| io::Error::other("Unknown endpoint"))?;
215        for packet in packets {
216            let len = packet.data.len();
217            match s.try_send(packet) {
218                Ok(_) => info!(
219                    "send {} -> {}: sent {} bytes",
220                    self.id.fmt_short(),
221                    to_id.fmt_short(),
222                    len
223                ),
224                Err(TrySendError::Full(_)) => info!(
225                    "send {} -> {}: dropped {} bytes",
226                    self.id.fmt_short(),
227                    to_id.fmt_short(),
228                    len
229                ),
230                Err(TrySendError::Closed(_)) => return Err(io::Error::other("channel closed")),
231            }
232        }
233        Ok(())
234    }
235
236    fn split(&self, transmit: &Transmit) -> impl Iterator<Item = Packet> {
237        let from = to_custom_addr(self.id);
238        let segment_size = transmit.segment_size.unwrap_or(transmit.contents.len());
239        transmit
240            .contents
241            .chunks(segment_size)
242            .map(move |slice| Packet {
243                from: from.clone(),
244                data: Bytes::copy_from_slice(slice),
245            })
246    }
247}

284    fn poll_recv(
285        &mut self,
286        cx: &mut std::task::Context,
287        bufs: &mut [io::IoSliceMut<'_>],
288        metas: &mut [noq_udp::RecvMeta],
289        recv_infos: &mut [RecvInfo],
290    ) -> Poll<io::Result<usize>> {
291        assert_eq!(bufs.len(), metas.len(), "non matching bufs & metas");
292        assert_eq!(
293            bufs.len(),
294            recv_infos.len(),
295            "non matching bufs & recv_infos"
296        );
297        let n = bufs.len();
298        if n == 0 {
299            return Poll::Ready(Ok(0));
300        }
301        let mut guard = self.network.inner.lock().expect("poisoned");
302        let Some((_, r)) = guard.channels.get_mut(&self.id) else {
303            info!("me: {} not found in channels", self.id.fmt_short());
304            return Poll::Ready(Ok(0));
305        };
306        let mut packets = Vec::new();
307        match r.poll_recv_many(cx, &mut packets, n) {
308            Poll::Pending => return Poll::Pending,
309            Poll::Ready(0) => return Poll::Ready(Err(io::Error::other("channel closed"))),
310            Poll::Ready(n) => n,
311        };
312        let mut count = 0;
313        for (i, packet) in packets.into_iter().enumerate() {
314            let meta = &mut metas[i];
315            let buf = &mut bufs[i];
316            let recv_info = &mut recv_infos[i];
317            if buf.len() < packet.data.len() {
318                break;
319            }
320            let from = try_parse_custom_addr(&packet.from).expect("valid custom addr");
321            info!(
322                "recv {} -> {}: copying {} bytes",
323                from.fmt_short(),
324                self.id.fmt_short(),
325                packet.data.len()
326            );
327            buf[..packet.data.len()].copy_from_slice(&packet.data);
328            *recv_info = RecvInfo::new(packet.from, Some(to_custom_addr(self.id)));
329            meta.len = packet.data.len();
330            meta.stride = packet.data.len();
331            count += 1;
332        }
333        if count > 0 {
334            info!("recv {}: filled {count} slots", self.id.fmt_short());
335            Poll::Ready(Ok(count))
336        } else {
337            Poll::Pending
338        }
339    }

402    /// Creates a basic endpoint builder with the given secret key and custom transport.
403    fn endpoint_builder(
404        secret_key: SecretKey,
405        transport: Arc<TestTransport>,
406        config: EndpointConfig,
407    ) -> Builder {
408        let relay_mode = match config.relay_map {
409            Some(map) => RelayMode::Custom(map),
410            None => RelayMode::Disabled,
411        };
412        let mut builder = Endpoint::builder(presets::N0)
413            .secret_key(secret_key)
414            .relay_mode(relay_mode)
415            .ca_tls_config(crate::tls::CaTlsConfig::insecure_skip_verify())
416            .add_custom_transport(transport);
417        if let Some(bias) = config.custom_bias {
418            builder = builder.path_selector(Arc::new(
419                BiasedRttPathSelector::default()
420                    .with_bias(AddrKind::Custom(TEST_TRANSPORT_ID), bias),
421            ));
422        }
423        if !config.keep_ip {
424            builder = builder.clear_ip_transports();
425        }
426        builder
427    }

451    /// Returns true if the selected path is the custom transport.
452    fn is_custom_selected(conn: &crate::endpoint::Connection) -> bool {
453        let paths = conn.paths();
454        paths.iter().find(|p| p.is_selected()).is_some_and(
455            |p| matches!(p.remote_addr(), TransportAddr::Custom(a) if a.id() == TEST_TRANSPORT_ID),
456        )
457    }

459    /// Returns true if either
460    /// - we have both IP and custom paths, and the selected path is IP.
461    /// - we only have one path
462    fn is_ip_selected_from_ip_and_custom(conn: &crate::endpoint::Connection) -> bool {
463        let paths = conn.paths();
464        let has_ip = paths.iter().any(|p| p.remote_addr().is_ip());
465        let has_custom = paths.iter().any(|p| p.remote_addr().is_custom());
466        if !has_ip || !has_custom {
467            return true;
468        }
469        paths
470            .iter()
471            .any(|p| p.is_selected() && p.remote_addr().is_ip())
472    }

474    /// Returns true if the selected path is a relay transport.
475    fn is_relay_selected(conn: &crate::endpoint::Connection) -> bool {
476        let paths = conn.paths();
477        paths
478            .iter()
479            .find(|p| p.is_selected())
480            .is_some_and(|p| p.is_relay())

485        let (mut send, mut recv) = conn.open_bi().await.anyerr()?;

673
674        // Get all addresses including relay and custom
675        let ep2_addr = ep2.addr();
676        let custom_addr = to_custom_addr(s2.public());

769    /// Regression test for the cmux dead-direct-path kill loop: when the
770    /// *selected* direct path goes dark while a healthy relay path is open on
771    /// the same connection, iroh must promptly demote it without closing QUIC.
772    ///
773    /// Field condition being modeled (see
774    /// out/iroh-fork-program/B1-mechanism-note.md in cmuxterm-hq): both a
775    /// direct (primary-tier) and a relay (backup-tier) path exist; the direct
776    /// path is selected (`PathStatus::Available`, relay `Backup`); the direct
777    /// path then black-holes. Because noq pins ALL SpaceKind::Data frames
778    /// (streams, retransmits, ACKs) to the validated+Available path, and iroh
779    /// without iroh's health detector application data stalls until the
780    /// per-path idle timeout (`PATH_MAX_IDLE_TIMEOUT` = 15s) abandons the dead
781    /// path — even though the relay could carry the data the whole time.
782    /// cmux's session layer kills the connection ~2.2s into that black hole
783    /// (sendQueueOverflow / control watchdog), producing the ~2s metronome
784    /// kill loop in the field.
785    ///
786    /// The contract is survival plus relay failover within a generous 6s CI
787    /// margin (the detector normally reacts in ~1-2s), followed by sustained
788    /// application traffic on the same connection.
789    #[tokio::test]
790    #[traced_test]
791    async fn test_dead_selected_direct_path_demotes_to_relay() -> Result<()> {
792        /// The normal ~1-2s demotion gets ample CI headroom while remaining
793        /// far below the old ~15s path-idle recovery.
794        const RECOVERY_DEADLINE: Duration = Duration::from_secs(6);
795        /// Keep probing well past the pre-fix ~15s path-idle recovery so a
796        /// regression is diagnosed as "slow recovery at ~15s" rather than
797        /// "permanently wedged".
798        const PROBE_BUDGET: Duration = Duration::from_secs(45);
799
800        let (relay_map, _relay_url, _guard) = run_relay_server().await?;
801        let network = TestNetwork::new();
802        let s1 = SecretKey::generate();
803        let s2 = SecretKey::generate();
804
805        let t1 = network.create_transport(s1.public())?;
806        let t2 = network.create_transport(s2.public())?;
807
808        // IP transports are cleared by default in `endpoint_builder`, so the
809        // custom transport is the only primary-tier ("direct") path and the
810        // relay is the only backup-tier path — the exact field topology.
811        let config = EndpointConfig::default().with_relay(relay_map.clone());
812        let ep1 = endpoint_builder(s1, t1, config.clone()).bind().await?;
813        let ep2 = endpoint_builder(s2.clone(), t2, config).bind().await?;
814
815        // Wait for both endpoints to have their relay connection.
816        ep1.online().await;
817        ep2.online().await;
818
819        let router = Router::builder(ep2.clone()).accept(ECHO_ALPN, Echo).spawn();
820
821        // Dial with BOTH the relay and the custom address so both paths open.
822        let ep2_addr = ep2.addr();
823        let all_addrs = EndpointAddr::from_parts(
824            s2.public(),
825            ep2_addr
826                .addrs
827                .iter()
828                .cloned()
829                .chain(std::iter::once(TransportAddr::Custom(to_custom_addr(
830                    s2.public(),
831                )))),
832        );
833        let conn = ep1.connect(all_addrs, ECHO_ALPN).await?;
834
835        // Wait until the steady state we are testing: the direct (custom)
836        // path is selected AND a relay path is open on the connection.
837        let settle_deadline = Instant::now() + Duration::from_secs(15);
838        loop {
839            let paths = conn.paths();
840            let has_relay = paths.iter().any(|p| p.is_relay());
841            if has_relay && is_custom_selected(&conn) {
842                break;
843            }
844            assert!(
845                Instant::now() < settle_deadline,
846                "test setup failed: custom-selected + relay path never settled; paths: {:?}",
847                paths
848                    .iter()
849                    .map(|p| format!("{} selected={}", p.remote_addr(), p.is_selected()))
850                    .collect::<Vec<_>>()
851            );
852            tokio::time::sleep(Duration::from_millis(50)).await;
853        }
854
855        // One long-lived echo stream; every probe is one byte round-tripped.
856        let (mut send, mut recv) = conn.open_bi().await.anyerr()?;
857
858        // Prove the stream echoes while the direct path is healthy.
859        send.write_all(&[0xA5]).await.anyerr()?;
860        let mut byte = [0u8; 1];
861        tokio::time::timeout(Duration::from_secs(5), recv.read_exact(&mut byte))
862            .await
863            .std_context("echo did not work before the direct path went dark")?
864            .anyerr()?;
865        assert_eq!(byte, [0xA5]);
866
867        // The selected direct path goes completely dark. The relay stays up.
868        network.set_blackhole(true);
869        let dark_at = Instant::now();
870
871        // Probe: write a byte roughly every 500ms and wait for any echo.
872        // Under the bug nothing comes back until noq abandons the dead path
873        // (~15s). Track when data first flows again, or whether the
874        // connection dies first.
875        let mut first_echo_after_dark: Option<Duration> = None;
876        let mut connection_error: Option<String> = None;
877        while dark_at.elapsed() < PROBE_BUDGET {
878            if let Err(err) = send.write_all(&[0x5A]).await {
879                connection_error = Some(format!("write failed: {err:#}"));
880                break;
881            }
882            match tokio::time::timeout(Duration::from_millis(500), recv.read_exact(&mut byte)).await
883            {
884                Ok(Ok(())) => {
885                    first_echo_after_dark = Some(dark_at.elapsed());
886                    break;
887                }
888                Ok(Err(err)) => {
889                    connection_error = Some(format!("read failed: {err:#}"));
890                    break;
891                }
892                Err(_) => continue, // still stalled, keep probing
893            }
894        }
895
896        eprintln!(
897            "dead-direct probe: first_echo_after_dark={first_echo_after_dark:?} \
898             connection_error={connection_error:?} close_reason={:?} paths={:?}",
899            conn.close_reason(),
900            conn.paths()
901                .iter()
902                .map(|p| format!("{} selected={}", p.remote_addr(), p.is_selected()))
903                .collect::<Vec<_>>()
904        );
905
906        assert_eq!(
907            connection_error, None,
908            "the connection errored while the selected direct path was dark"
909        );
910        assert_eq!(
911            conn.close_reason(),
912            None,
913            "demoting the selected path must not close the QUIC connection"
914        );
915        let stall = first_echo_after_dark
916            .expect("application traffic never recovered over the open relay path");
917        assert!(
918            stall < RECOVERY_DEADLINE,
919            "relay recovery took {stall:?}, expected less than {RECOVERY_DEADLINE:?} \
920             (the old path-idle behavior took ~15s)"
921        );
922        assert!(
923            is_relay_selected(&conn),
924            "the relay path should be selected after the dead direct path is demoted"
925        );
926
927        // The echo handler serves this one long-lived stream. Earlier one-byte
928        // probes may already be queued, so read through them until the two new
929        // sentinel bytes prove sustained traffic after recovery.
930        send.write_all(&[0xB6, 0xC7]).await.anyerr()?;
931        tokio::time::timeout(Duration::from_secs(5), async {
932            loop {
933                recv.read_exact(&mut byte).await?;
934                if byte == [0xB6] {
935                    recv.read_exact(&mut byte).await?;
936                    assert_eq!(byte, [0xC7]);
937                    return std::result::Result::<(), noq::ReadExactError>::Ok(());
938                }
939                assert_eq!(byte, [0x5A]);
940            }
941        })
942        .await
943        .std_context("sustained echo did not work after relay recovery")?
944        .anyerr()?;
945
946        conn.close(0u32.into(), b"done");
947        router.shutdown().await.anyerr()?;
948        Ok(())
949    }

951    /// Regression test: keepalive PINGs and their PTO probe retransmissions
952    /// must never count as stall evidence for the selected-path health
953    /// detector.
954    ///
955    /// Field condition being modeled: an established connection sits idle (no
956    /// application data pending) on a selected direct path when the radio
957    /// drops for a few seconds — a WiFi roam or channel switch, well below
958    /// `PATH_MAX_IDLE_TIMEOUT` (15s), which the idle timeout was deliberately
959    /// sized to tolerate. The only traffic in that gap is the 5s keepalive
960    /// PING (`HEARTBEAT_INTERVAL`) plus its PTO probe retransmissions. If the
961    /// detector counts those raw datagrams as stall evidence, one lost
962    /// keepalive and two probes reach the demotion threshold within the 1s
963    /// stall floor, and the healthy direct path is demoted AND quarantined
964    /// (5s doubling to 300s) with zero application data affected. Repeated
965    /// transients then ratchet the quarantine and pin the user to the relay.
966    #[tokio::test]
967    #[traced_test]
968    async fn test_selected_direct_path_idle_transient_loss_does_not_demote() -> Result<()> {
969        /// Longer than one `HEARTBEAT_INTERVAL` (5s), so at least one
970        /// keepalive PING is lost and PTO-probed inside the gap, while
971        /// staying well below `PATH_MAX_IDLE_TIMEOUT` (15s), which must be
972        /// the only mechanism allowed to abandon an idle path.
973        const DARK_WINDOW: Duration = Duration::from_secs(8);
974
975        let (relay_map, _relay_url, _guard) = run_relay_server().await?;
976        let network = TestNetwork::new();
977        let s1 = SecretKey::generate();
978        let s2 = SecretKey::generate();
979
980        let t1 = network.create_transport(s1.public())?;
981        let t2 = network.create_transport(s2.public())?;
982
983        // Same topology as the dead-path demotion test: the custom transport
984        // is the only primary-tier ("direct") path, the relay the only
985        // backup-tier path. The relay's presence is what makes a wrong
986        // demotion possible at all (`has_alternative`).
987        let config = EndpointConfig::default().with_relay(relay_map.clone());
988        let ep1 = endpoint_builder(s1, t1, config.clone()).bind().await?;
989        let ep2 = endpoint_builder(s2.clone(), t2, config).bind().await?;
990
991        ep1.online().await;
992        ep2.online().await;
993
994        let router = Router::builder(ep2.clone()).accept(ECHO_ALPN, Echo).spawn();
995
996        let ep2_addr = ep2.addr();
997        let all_addrs = EndpointAddr::from_parts(
998            s2.public(),
999            ep2_addr
1000                .addrs
1001                .iter()
1002                .cloned()
1003                .chain(std::iter::once(TransportAddr::Custom(to_custom_addr(
1004                    s2.public(),
1005                )))),
1006        );
1007        let conn = ep1.connect(all_addrs, ECHO_ALPN).await?;
1008
1009        // Wait for the steady state: direct (custom) selected, relay open.
1010        let settle_deadline = Instant::now() + Duration::from_secs(15);
1011        loop {
1012            let paths = conn.paths();
1013            let has_relay = paths.iter().any(|p| p.is_relay());
1014            if has_relay && is_custom_selected(&conn) {
1015                break;
1016            }
1017            assert!(
1018                Instant::now() < settle_deadline,
1019                "test setup failed: custom-selected + relay path never settled; paths: {:?}",
1020                paths
1021                    .iter()
1022                    .map(|p| format!("{} selected={}", p.remote_addr(), p.is_selected()))
1023                    .collect::<Vec<_>>()
1024            );
1025            tokio::time::sleep(Duration::from_millis(50)).await;
1026        }
1027
1028        // One long-lived echo stream (the Echo handler serves exactly one).
1029        let (mut send, mut recv) = conn.open_bi().await.anyerr()?;
1030        let mut byte = [0u8; 1];
1031
1032        // Prove the stream echoes while the direct path is healthy.
1033        send.write_all(&[0xA5]).await.anyerr()?;
1034        tokio::time::timeout(Duration::from_secs(5), recv.read_exact(&mut byte))
1035            .await
1036            .std_context("echo did not work before the transient loss")?
1037            .anyerr()?;
1038        assert_eq!(byte, [0xA5]);
1039
1040        // Let the echo's trailing retransmits/ACKs drain so the connection is
1041        // genuinely idle: no application data pending anywhere.
1042        tokio::time::sleep(Duration::from_secs(1)).await;
1043
1044        // Transient radio gap: every packet on the direct transport is lost,
1045        // while the connection stays idle. Only keepalives and PTO probes are
1046        // transmitted during this window.
1047        network.set_blackhole(true);
1048        let dark_at = Instant::now();
1049        while dark_at.elapsed() < DARK_WINDOW {
1050            assert_eq!(
1051                conn.close_reason(),
1052                None,
1053                "the connection must survive an idle transient loss"
1054            );
1055            assert!(
1056                is_custom_selected(&conn),
1057                "idle transient loss demoted the selected direct path after {:?} \
1058                 (only keepalive/PTO traffic was in flight); paths: {:?}",
1059                dark_at.elapsed(),
1060                conn.paths()
1061                    .iter()
1062                    .map(|p| format!("{} selected={}", p.remote_addr(), p.is_selected()))
1063                    .collect::<Vec<_>>()
1064            );
1065            tokio::time::sleep(Duration::from_millis(250)).await;
1066        }
1067        network.set_blackhole(false);
1068
1069        // Give the next keepalive/PTO probe a moment to round-trip now that
1070        // the transient cleared.
1071        tokio::time::sleep(Duration::from_millis(500)).await;
1072
1073        // The path was never demoted, so it must still be selected (a
1074        // demotion would have quarantined the direct remote for >= 5s,
1075        // pinning selection to the relay well past this point).
1076        assert!(
1077            is_custom_selected(&conn),
1078            "the direct path must still be selected after the transient cleared; paths: {:?}",
1079            conn.paths()
1080                .iter()
1081                .map(|p| format!("{} selected={}", p.remote_addr(), p.is_selected()))
1082                .collect::<Vec<_>>()
1083        );
1084
1085        // And it must still carry application data on the same stream.
1086        send.write_all(&[0x42]).await.anyerr()?;
1087        tokio::time::timeout(Duration::from_secs(5), recv.read_exact(&mut byte))
1088            .await
1089            .std_context("echo did not work after the transient loss cleared")?
1090            .anyerr()?;
1091        assert_eq!(byte, [0x42]);
1092        assert!(
1093            is_custom_selected(&conn),
1094            "application traffic after the transient must still ride the direct path"
1095        );
1096
1097        conn.close(0u32.into(), b"done");
1098        router.shutdown().await.anyerr()?;
1099        Ok(())
1100    }
